cmake_minimum_required(VERSION 3.13)
set(PROJECT_AND_EXECATUBLE_NAME SDL_Engine_Base)
project(${PROJECT_AND_EXECATUBLE_NAME})
set(CMAKE_CXX_STANDARD 17)

option(USE_SDL2 "Use SDL2 Renderer" OFF)
option(USE_OPENGL "Use OpenGL Renderer" ON)
option(USE_VULKAN "Use Vulkan Renderer" OFF) # NOT USED YET
option(USE_OPENGL_ES "Use SDL2 Renderer" OFF)
option(EMSCRIPTEN_BUILD "Is Emscripten Build" OFF)
option(LINUX_BUILD "Is Linux Build" OFF)

if (UNIX)
    if (CMAKE_CXX_COMPILER MATCHES "/em\\+\\+(-[a-zA-Z0-9.])?$")
        set(EMSCRIPTEN_BUILD ON)
        set(USE_OPENGL_ES ON)
        set(USE_VULKAN OFF)
        set(USE_OPENGL OFF)
    else ()
        set(LINUX_BUILD ON)
    endif ()
endif ()

if (USE_SDL2D)
    message("Using SDL2D Renderer")
    add_compile_definitions(USE_SDL2D)
    file(GLOB_RECURSE RENDERER_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Engine/Rendering/2D/*.c*")
elseif (USE_OPENGL)
    message("Use OpenGL")
    add_compile_definitions(USE_OPENGL)
    file(GLOB_RECURSE RENDERER_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Engine/Rendering/OpenGL/*.c*")
elseif (USE_OPENGL_ES)
    message("Use OpenGL ES")
    add_compile_definitions(USE_OPENGL_ES)
    
    file(GLOB_RECURSE GL_SOURCE "${CMAKE_SOURCE_DIR}/src/Engine/Rendering/OpenGL/*.c*")
    file(GLOB_RECURSE GL_ES_SOURCE "${CMAKE_SOURCE_DIR}/src/Engine/Rendering/OpenGL_ES/*.c*")
    set(GL_ES_BASENAMES)
    foreach (GL_ES_FILE ${GL_ES_SOURCE})
        get_filename_component(BASENAME ${GL_ES_FILE} NAME)
        list (APPEND GL_ES_BASENAMES ${BASENAME})
    endforeach ()

    set(FILTERED_GL_FILES)
    foreach (GL_FILE ${GL_SOURCE})
        get_filename_component(BASANAME ${GL_FILE} NAME)
        if (NOT BASENAME IN_LIST GL_ES_BASENAMES)
            list(APPEND FILTERED_GL_FILES ${GL_FILE})
        endif ()
    endforeach ()

    set (RENDERER_SOURCE_FILES ${FILTERED_GL_FILES} ${GL_ES_SOURCE})
elseif (USE_VULKAN)
    #NOT USED NOW
endif()

file(GLOB_RECURSE AUDIO_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Engine/Audio/*.c*")
file(GLOB_RECURSE MATH_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Engine/Math/*.c*")
file(GLOB_RECURSE HELPER_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Engine/Helper/*.c*")
file(GLOB ENGINE_BASE_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Engine/*.c*")

set (ENGINE_SOURCE_FILES ${ENGINE_BASE_SOURCE_FILES} ${HELPER_SOURCE_FILES} ${MATH_SOURCE_FILES} ${AUDIO_SOURCE_FILES} ${RENDERER_SOURCE_FILES})
set (GAME_SOURCE_FILES "${CMAKE_SOURCE_DIR}/src/Game.cpp")
set (SOURCE_FILES ${ENGINE_SOURCE_FILES} ${GAME_SOURCE_FILES} "${CMAKE_SOURCE_DIR}/src/Main.cpp")

message("Source Files List:")
foreach(SOURCE_FILE ${SOURCE_FILES})
    message("${SOURCE_FILE}")
endforeach()

if (EMSCRIPTEN_BUILD)
    message("Emscripten")
    set(CMAKE_CXX_COMPILER_ID "Emscripten")
    set(CMAKE_EXECUTABLE_SUFFIX ".html")
    
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -sFULL_ES3 -sMIN_WEBGL_VERSION=1 -sMAX_WEBGL_VERSION=2 --preload-file Assets")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -s USE_SDL=2 -s USE_SDL_TTF=2 -s USE_SDL_NET=2 -s USE_SDL_IMAGE=2 -sUSE_SDL_MIXER=2 -s SDL2_IMAGE_FORMATS=[\"png\"]")

    include_directories(${SDL2_INCLUDE_DIRS} ${SDL2_IMAGE_INCLUDE_DIRS} ${SDL2_MIXER_INCLUDE_DIRS} ${SDL2_net_INCLUDE_DIRS} ${SDL2_TTF_INCLUDE_DIR})
    add_executable(${PROJECT_AND_EXECATUBLE_NAME} ${SOURCE_FILES})
    target_link_libraries(${PROJECT_AND_EXECATUBLE_NAME} PRIVATE ${SDL2_LIBRARIES} ${SDL2_IMAGE_LIBRARIES} ${SDL2_MIXER_LIBRARIES} ${SDL2_TTF_LIBRARIES})
elseif (UNIX)
    message("UNIX")

    set(CMAKE_MODULE_PATH ${CMAKE_MODULE_PATH} "${CMAKE_SOURCE_DIR}/cmake/linux/")
    find_package(SDL2 REQUIRED)
    find_package(SDL2_image REQUIRED)
    find_package(SDL2_mixer REQUIRED)
    find_package(SDL2_ttf REQUIRED)
    find_package(SDL2_net REQUIRED)
    find_package(OpenGL REQUIRED)

    include_directories(${SDL2_INCLUDE_DIRS} ${SDL2_IMAGE_INCLUDE_DIRS} ${SDL2_MIXER_INCLUDE_DIRS} ${SDL2_net_INCLUDE_DIRS} ${SDL2_TTF_INCLUDE_DIR})
    add_executable(${PROJECT_AND_EXECATUBLE_NAME} ${SOURCE_FILES})
    target_link_libraries(${PROJECT_AND_EXECATUBLE_NAME} PRIVATE ${SDL2_LIBRARIES} ${SDL2_IMAGE_LIBRARIES} ${SDL2_MIXER_LIBRARIES} ${SDL2_TTF_LIBRARIES})
elseif (WIN32)
  message("WINDOWS")
  find_package(SDL2 CONFIG REQUIRED)
  find_package(SDL2_image CONFIG REQUIRED)
  find_package(SDL2_mixer CONFIG REQUIRED)
  find_package(SDL2_ttf CONFIG REQUIRED)
  find_package(sdl2_net CONFIG REQUIRED)
  find_package(glm CONFIG REQUIRED)

  include_directories(${SDL2_INCLUDE_DIRS} ${SDL2_IMAGE_INCLUDE_DIRS} ${SDL2_MIXER_INCLUDE_DIRS} ${SDL2_net_INCLUDE_DIRS} ${SDL2_TTF_INCLUDE_DIR} )
  add_executable(${PROJECT_AND_EXECATUBLE_NAME} ${SOURCE_FILES})
  target_link_libraries(${PROJECT_AND_EXECATUBLE_NAME} PRIVATE SDL2::SDL2 SDL2::SDL2main SDL2_image::SDL2_image SDL2_mixer::SDL2_mixer SDL2_net::SDL2_net SDL2_ttf::SDL2_ttf glm::glm)
endif ()
